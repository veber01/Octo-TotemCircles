cmake_minimum_required(VERSION 3.21)
project(TotemCircles LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "WoW 1.12.1 needs an x86 / Win32 DLL. Configure with -A Win32")
endif()
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4)
FetchContent_MakeAvailable(minhook)
add_library(TotemCircles SHARED src/TotemCircles.cpp src/TotemCircles.def)
target_compile_features(TotemCircles PRIVATE cxx_std_17)
target_link_libraries(TotemCircles PRIVATE minhook user32 d3d9)
target_compile_definitions(TotemCircles PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(TotemCircles PROPERTIES OUTPUT_NAME "TotemCircles")
if(MSVC)
  target_compile_options(TotemCircles PRIVATE /W4 /EHsc)
endif()
